cmake_minimum_required(VERSION 3.16)
project(TohoCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(toho_core game.cpp environment.cpp)
target_include_directories(toho_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(toho_core_tests test_game.cpp)
target_link_libraries(toho_core_tests PRIVATE toho_core)
enable_testing()
add_test(NAME gameplay COMMAND toho_core_tests)
add_executable(toho_environment_tests test_environment.cpp)
target_link_libraries(toho_environment_tests PRIVATE toho_core)
add_test(NAME environment COMMAND toho_environment_tests)
